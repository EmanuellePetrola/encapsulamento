class Program
{
    static void Main(string[] args)
    {
        Carro carro = new Carro();

        carro.Acelerar();
        Console.WriteLine("Velocidade: " + carro.VelocidadeAtual + " km/h");

        carro.Acelerar();
        Console.WriteLine("Velocidade: " + carro.VelocidadeAtual + " km/h");

        carro.Acelerar();
        Console.WriteLine("Velocidade: " + carro.VelocidadeAtual + " km/h");

        carro.Frear();
        Console.WriteLine("Velocidade: " + carro.VelocidadeAtual + " km/h");

        carro.Frear();
        Console.WriteLine("Velocidade: " + carro.VelocidadeAtual + " km/h");

        carro.VelocidadeAtual = 100;
    }
}
class Carro
{
    private int _velocidadeAtual;

    public int VelocidadeAtual
    {
        get { return _velocidadeAtual; }
    }

    public void Acelerar()
    {
        _velocidadeAtual += 10;
    }

    public void Frear()
    {
        if (_velocidadeAtual >= 10)
        {
            _velocidadeAtual -= 10;
        }
        else
        {
            _velocidadeAtual = 0;
        }
    }
}
